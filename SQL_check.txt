-- fetch 
select max(salary) from EMPLOYEE;
select max(salary) from EMPLOYEE where salary < (select max(salary) from EMPLOYEE );
select max(salary) from EMPLOYEE where salary < (select max(salary) from EMPLOYEE where salary < (select max(salary) from EMPLOYEE));


-- to get the count with group by
select count(*),avg(salary) as average_salary from EMPLOYEE  group by dept;
select count(*),avg(salary) from EMPLOYEE  group by dept having count(*)>1;
-- to find number of EMPLOYEE in each dept
select dept,count(*) from EMPLOYEE group by dept



-- to get salary of greatern then the by dep average
SELECT e.salary
FROM EMPLOYEE e
WHERE e.salary > (
    SELECT AVG(f.salary)
    FROM EMPLOYEE f 
    where e.dept = f.dept );


-- to select top 3 salary
select distinct salary from EMPLOYEE order by salary desc fetch first 3 rows only;
-- or
select distinct salary from EMPLOYEE order by salary desc limit 3;

-- with join
select e.empId e.name from EMPLOYEE e left join project p on e.empId = p.empId

-- to rank EMPLOYEE based on salary
select empId,name,salary,rank() over(order by salary desc) from EMPLOYEE;
-- by dept
select empId,name,salary,rank() over(
  PARTITION BY dept  
  order by salary desc ) 
from EMPLOYEE
